<script>
  //==================== GLOBAL STATE ====================
let currentUser = null;
let currentPage = 'dashboard';
let dashboardData = null;
let dashboardEmployees = [];
let masterDataType = '';
let editingId = null;
let currentCoachingId = null;

//==================== INITIALIZATION ====================
document.addEventListener('DOMContentLoaded', function() {
  console.log('DOM Loaded - Starting app v4.0...');
  
  if (typeof google === 'undefined' || !google.script || !google.script.run) {
    console.error('Google Apps Script environment not detected');
    showToast('error', 'Error', 'Aplikasi harus dijalankan dari Google Apps Script');
    hideLoading();
    return;
  }
  
  const timeoutId = setTimeout(function() {
    console.log('Session check timeout - showing login');
    hideLoading();
  }, 5000);
  
  checkSession(function() {
    clearTimeout(timeoutId);
  });
  
  initWeeksDropdown();
  initReminderListeners();
});

function hideLoading() {
  const loadingScreen = document.getElementById('loadingScreen');
  if (loadingScreen) {
    loadingScreen.classList.add('hidden');
  }
}

function showLoading() {
  const loadingScreen = document.getElementById('loadingScreen');
  if (loadingScreen) {
    loadingScreen.classList.remove('hidden');
  }
}

function initReminderListeners() {
  const reminderTime = document.getElementById('reminderTime');
  if (reminderTime) {
    reminderTime.addEventListener('change', function() {
      const customGroup = document.getElementById('customDateGroup');
      if (this.value === 'custom') {
        customGroup.style.display = 'block';
      } else {
        customGroup.style.display = 'none';
      }
    });
  }
}

//==================== AUTHENTICATION ====================
function checkSession(callback) {
  console.log('Checking session...');
  try {
    google.script.run
      .withSuccessHandler(function(result) {
        console.log('Session check result:', result);
        if (result && result.npk) {
          currentUser = result;
          showApp();
        } else {
          hideLoading();
        }
        if (callback) callback();
      })
      .withFailureHandler(function(error) {
        console.error('Session check error:', error);
        hideLoading();
        if (callback) callback();
      })

function animateNumber(id, value) {
  const element = document.getElementById(id);
  if (!element) return;
  const start = parseInt(element.textContent) || 0;
  const end = value;
  const duration = 800;
  const startTime = performance.now();
  
  function update(currentTime) {
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);
    const easeProgress = 1 - Math.pow(1 - progress, 3);
    const current = Math.floor(start + (end - start) * easeProgress);
    element.textContent = current;
    if (progress < 1) requestAnimationFrame(update);
  }
  requestAnimationFrame(update);
}

// SORTING FUNCTION
function sortTable(column) {
  if (currentSort.column === column) {
    currentSort.direction = currentSort.direction === 'asc' ? 'desc' : 'asc';
  } else {
    currentSort.column = column;
    currentSort.direction = 'asc';
  }
  
  // Update header icons
  document.querySelectorAll('#dashboardEmployeeTable th').forEach(th => {
    th.classList.remove('sort-asc', 'sort-desc');
  });
  
  const th = document.querySelector(`#dashboardEmployeeTable th[onclick="sortTable('${column}')"]`);
  if (th) {
    th.classList.add(currentSort.direction === 'asc' ? 'sort-asc' : 'sort-desc');
  }
  
  // Sort data
  const sorted = [...dashboardEmployees].sort((a, b) => {
    let valA = a[column] || '';
    let valB = b[column] || '';
    
    // Special handling for status
    if (column === 'status') {
      const statusOrder = { 'BELUM_COACHING': 0, 'BELUM_UPDATE': 1, 'SUDAH_UPDATE': 2 };
      valA = statusOrder[valA] || 0;
      valB = statusOrder[valB] || 0;
    }
    
    if (typeof valA === 'string') valA = valA.toLowerCase();
    if (typeof valB === 'string') valB = valB.toLowerCase();
    
    if (valA < valB) return currentSort.direction === 'asc' ? -1 : 1;
    if (valA > valB) return currentSort.direction === 'asc' ? 1 : -1;
    return 0;
  });
  
  renderDashboardEmployees(sorted);
}

function initWeeksDropdown() {
  const select = document.getElementById('inputWeek');
  if (!select) return;
  
  const currentYear = new Date().getFullYear();
  const currentWeek = getCurrentISOWeek();
  
  select.innerHTML = '';
  for (let i = 1; i <= 52; i++) {
    const week = `${currentYear}-W${i.toString().padStart(2, '0')}`;
    const option = document.createElement('option');
    option.value = week;
    option.textContent = week;
    if (week === currentWeek) option.selected = true;
    select.appendChild(option);
  }
}

function getCurrentISOWeek() {
  const now = new Date();
  const d = new Date(Date.UTC(now.getFullYear(), now.getMonth(), now.getDate()));
  const dayNum = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  const weekNo = Math.ceil((((d - yearStart) / 86400000) + 1) / 7);
  return `${d.getUTCFullYear()}-W${weekNo.toString().padStart(2, '0')}`;
}

  const coacheeNpk = document.getElementById('inputCoachee').value;
  if (!coacheeNpk) return;
  
  try {
    google.script.run
      .withSuccessHandler(function(result) {
        if (result && result.success) {
          if (result.hasActive) {
            // Show update form
            showUpdateForm(result.coaching);
          } else {
            // Show header form
            showHeaderForm();
          }
        }
      })
      .getActiveCoachingForCoachee(coacheeNpk);
  } catch (e) {
    console.error('Exception in checkActiveCoaching:', e);
  }
}

function showUpdateForm(coaching) {
  document.getElementById('formStateHeader').classList.remove('active');
  document.getElementById('formStateUpdate').classList.add('active');
  document.getElementById('activeCoachingBanner').classList.remove('hidden');
  
  document.getElementById('updateCoachingId').value = coaching.coaching_id;
  document.getElementById('activeCoachingText').innerHTML = `
    <strong>${coaching.root_cause}</strong><br>
    Topic: ${coaching.topic}<br>
    Status: <span class="badge ${coaching.status === 'OVERDUE' ? 'badge-danger' : 'badge-warning'}">${coaching.status}</span>
  `;
  
  // Pre-fill cabang if empty
  if (!document.getElementById('inputCabang').value) {
    document.getElementById('inputCabang').value = coaching.cabang;
  }
}

function showHeaderForm() {
  document.getElementById('formStateHeader').classList.add('active');
  document.getElementById('formStateUpdate').classList.remove('active');
  document.getElementById('activeCoachingBanner').classList.add('hidden');
  document.getElementById('updateCoachingId').value = '';

function switchToUpdateMode() {

function saveReminder() {
  const timeOption = document.getElementById('reminderTime').value;
  const useEmail = document.getElementById('reminderEmail').checked;
  const useWA = document.getElementById('reminderWA').checked;
  
  if (!useEmail && !useWA) {
    showToast('warning', 'Peringatan', 'Pilih minimal satu channel (Email atau WhatsApp)');
    return;
  }
  
  let reminderDate;
  const now = new Date();
  
  switch(timeOption) {
    case '24h':
      reminderDate = new Date(now.getTime() + 24 * 60 * 60 * 1000);
      break;
    case '3d':
      reminderDate = new Date(now.getTime() + 3 * 24 * 60 * 60 * 1000);
      break;
    case '7d':
      reminderDate = new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000);
      break;
    case 'custom':
      reminderDate = new Date(document.getElementById('customReminderDate').value);
      break;
  }
  
  const channel = [];
  if (useEmail) channel.push('email');
  if (useWA) channel.push('whatsapp');
  
  const data = {
    coaching_id: lastSavedCoaching?.coaching_id || currentCoachingId,
    coachee_npk: document.getElementById('inputCoachee')?.value || '',
    reminder_date: reminderDate.toISOString(),
    channel: channel.join(','),
    message: `Reminder: Follow-up coaching untuk ${document.getElementById('inputCoachee')?.selectedOptions?.[0]?.text || 'karyawan'}`
  };
  
  showLoading();
  try {
    google.script.run
      .withSuccessHandler(function(result) {
        hideLoading();
        closeReminderModal();
        if (result && result.success) {
          showToast('success', 'Reminder Berhasil', result.message);
        } else {
          showToast('error', 'Error', result ? result.message : 'Gagal membuat reminder');
        }
      })
      .withFailureHandler(function(error) {
        hideLoading();
        showToast('error', 'Error', 'Gagal membuat reminder: ' + error.message);
      })
      .createReminder(data);
  } catch (e) {
    hideLoading();
    showToast('error', 'Error', 'Terjadi kesalahan sistem');
  }
}

let masterSort = { type: '', column: '', direction: 'asc' };

function renderMasterTable(type, data) {
  const tbody = document.getElementById(`${type}TableBody`);
  if (!tbody) return;
  
  tbody.innerHTML = (data || []).map(item => `
    <tr>
      <td><strong>${item.npk || '-'}</strong></td>
      <td>${item.nama || '-'}</td>
      <td>${item.jabatan || '-'}</td>
      <td><span class="badge badge-primary">CL ${item.cl || '-'}</span></td>
      <td>${item.cabang || '-'}</td>
      <td>${item.region || '-'}</td>
      ${type === 'karyawan' ? `<td>${item.atasan_npk || '-'}</td>` : ''}
      <td>
        <div class="table-actions">
          <button class="action-btn edit" onclick="edit${type === 'user' ? 'User' : 'Karyawan'}('${item.npk}')" title="Edit">
            <span class="material-icons-round">edit</span>
          </button>
          <button class="action-btn delete" onclick="delete${type === 'user' ? 'User' : 'Karyawan'}('${item.npk}')" title="Hapus">
            <span class="material-icons-round">delete</span>
          </button>
        </div>
      </td>
    </tr>
  `).join('');
}

function sortMasterTable(type, column) {
  if (masterSort.column === column && masterSort.type === type) {
    masterSort.direction = masterSort.direction === 'asc' ? 'desc' : 'asc';
  } else {
    masterSort.type = type;
    masterSort.column = column;
    masterSort.direction = 'asc';
  }
  
  // Update header icons
  document.querySelectorAll(`#${type}Table th`).forEach(th => {
    th.classList.remove('sort-asc', 'sort-desc');
  });
  
  const th = document.querySelector(`#${type}Table th[onclick="sortMasterTable('${type}', '${column}')"]`);
  if (th) {
    th.classList.add(masterSort.direction === 'asc' ? 'sort-asc' : 'sort-desc');
  }
  
  // Get current data and sort
  const tbody = document.getElementById(`${type}TableBody`);
  const rows = Array.from(tbody.querySelectorAll('tr'));
  
  rows.sort((a, b) => {
    const valA = a.cells[getColumnIndex(type, column)].textContent.toLowerCase();
    const valB = b.cells[getColumnIndex(type, column)].textContent.toLowerCase();
    
    if (valA < valB) return masterSort.direction === 'asc' ? -1 : 1;
    if (valA > valB) return masterSort.direction === 'asc' ? 1 : -1;
    return 0;
  });
  
  tbody.innerHTML = '';
  rows.forEach(row => tbody.appendChild(row));
}

function getColumnIndex(type, column) {
  const columns = {
    'user': ['npk', 'nama', 'jabatan', 'cl', 'cabang', 'region'],
    'karyawan': ['npk', 'nama', 'jabatan', 'cl', 'cabang', 'region', 'atasan_npk']
  };
  return columns[type].indexOf(column);
}

function searchMasterData(type) {
  const query = document.getElementById(`search${type === 'user' ? 'User' : 'Karyawan'}`).value.toLowerCase();
  const rows = document.querySelectorAll(`#${type}TableBody tr`);
  
  rows.forEach(row => {
    const text = row.textContent.toLowerCase();
    row.style.display = text.includes(query) ? '' : 'none';
  });
}

//==================== UTILITY ====================
function showToast(type, title, message) {
  const container = document.getElementById('toastContainer');
  if (!container) return;
  
  const toast = document.createElement('div');
  toast.className = `toast ${type}`;
  
  const icons = {
    'success': 'check_circle',
    'error': 'error',
    'warning': 'warning',
    'info': 'info'
  };
  
  toast.innerHTML = `
    <div class="toast-icon">
      <span class="material-icons-round">${icons[type]}</span>
    </div>
    <div class="toast-content">
      <h4>${title}</h4>
      <p>${message}</p>
    </div>
    <button class="toast-close" onclick="this.parentElement.remove()">
      <span class="material-icons-round">close</span>
    </button>
  `;
  
  container.appendChild(toast);
  
  setTimeout(() => {
    toast.style.opacity = '0';
    toast.style.transform = 'translateX(100%)';
    setTimeout(() => toast.remove(), 300);
  }, 5000);
}

function formatDate(dateStr) {
  if (!dateStr) return '-';
  try {
    const date = new Date(dateStr);
    return date.toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' });
  } catch (e) {
    return dateStr;
  }
}

// Fallback: Load tanpa coaching status
function loadDashboardFallback() {
  console.log('Using fallback...');
  
  try {
    google.script.run
      .withSuccessHandler(function(result) {
        hideLoading();
        console.log('Fallback result:', result);
        
        if (result.success && result.employees) {
          // Render tanpa status coaching
          const simpleEmployees = result.employees.map(emp => ({
            ...emp,
            status: 'UNKNOWN',
            coaching_id: null
          }));
          
          dashboardEmployees = simpleEmployees;
          renderDashboardEmployees(simpleEmployees);
          
          showToast('warning', 'Mode Fallback', 'Menampilkan data dasar tanpa status coaching');
        }
      })
      .withFailureHandler(function(error) {
        hideLoading();
        showToast('error', 'Fatal Error', 'Tidak dapat memuat data: ' + error.message);
      })
      .getSimpleEmployeeList();
      
  } catch (e) {
    hideLoading();
    showToast('error', 'Error', 'Fallback gagal');
  }
}
